    @Override
    public void onInitialize() {
        loadConfig();

        ServerLifecycleEvents.SERVER_STARTED.register(s -> {
            server = s;
            if (cfg.startStop) say("✅ **" + esc(cfg.serverName) + "** ist online.");
        });
        ServerLifecycleEvents.SERVER_STOPPING.register(s -> {
            if (cfg.startStop) say("⛔ **" + esc(cfg.serverName) + "** wird gestoppt.");
            flushDiamonds();
            tick();
        });
        ServerPlayConnectionEvents.JOIN.register((handler, sender, s) -> {
            int n = ONLINE.incrementAndGet();
            if (cfg.join) say("🟢 **" + esc(Compat.playerName(handler)) + "** ist beigetreten (" + n + " online)");
            if (wants("join")) requestRefresh();
        });
        ServerPlayConnectionEvents.DISCONNECT.register((handler, s) -> {
            int n = ONLINE.updateAndGet(v -> Math.max(0, v - 1));
            if (cfg.leave) say("🔴 **" + esc(Compat.playerName(handler)) + "** hat den Server verlassen (" + n + " online)");
            if (wants("leave")) requestRefresh();
        });
        ServerMessageEvents.GAME_MESSAGE.register((s, message, overlay) -> {
            if (overlay) return;
            String key = Compat.key(message);
            if (key.startsWith("death.")) {
                if (cfg.death) say("💀 " + esc(message.getString()));
                if (wants("death")) requestRefresh();
            } else if (key.startsWith("chat.type.advancement.")) {
                if (cfg.advancement) say("🏆 " + esc(message.getString()));
                if (wants("advancement")) requestRefresh();
            }
        });
        PlayerBlockBreakEvents.AFTER.register((world, player, pos, state, be) -> {
            if (cfg.diamonds && Compat.isDiamond(state)) countDiamond(player.getName().getString());
        });

        if (cfg.commands) {
            Compat.registerCommands(LiveStats::command);
            EX.scheduleWithFixedDelay(LiveStats::loadSite, 5, 120, TimeUnit.SECONDS);
        }
        EX.scheduleWithFixedDelay(LiveStats::tick, 2, 2, TimeUnit.SECONDS);
        EX.scheduleWithFixedDelay(LiveStats::periodic, 60, 60, TimeUnit.SECONDS);
        LOG.info("LiveStats geladen (Discord: {}, Seiten-Aktualisierung: {})",
                cfg.discordWebhook.isBlank() ? "aus" : "an", cfg.githubToken.isBlank() ? "aus" : "an");
    }

    static String siteUrl() {
        String u = cfg.siteUrl == null ? "" : cfg.siteUrl.trim();
        if (u.isEmpty() && cfg.githubRepo != null && cfg.githubRepo.contains("/")) {
            String[] p = cfg.githubRepo.trim().split("/");
            u = "https://" + p[0].toLowerCase() + ".github.io/" + p[1] + "/";
        }
        return u.isEmpty() || u.endsWith("/") ? u : u + "/";
    }

    static String httpGet(String url) throws Exception {
        HttpResponse<String> r = HTTP.send(HttpRequest.newBuilder(URI.create(url + "?t=" + System.currentTimeMillis()))
                .timeout(Duration.ofSeconds(20)).header("User-Agent", "livestats").GET().build(), HttpResponse.BodyHandlers.ofString());
        if (r.statusCode() != 200) throw new IllegalStateException("HTTP " + r.statusCode() + " für " + url);
        return r.body();
    }

    static void loadSite() {
        String base = siteUrl();
        if (base.isEmpty()) return;
        try {
            if (siteKey == null) {
                if (cfg.serverKey != null && !cfg.serverKey.isBlank()) siteKey = cfg.serverKey.trim();
                else {
                    JsonArray a = JsonParser.parseString(httpGet(base + "servers.json")).getAsJsonArray();
                    for (JsonElement e : a) {
                        JsonObject o = e.getAsJsonObject();
                        if (o.get("name").getAsString().trim().equalsIgnoreCase(cfg.serverName.trim())) siteKey = o.get("key").getAsString();
                    }
                    if (siteKey == null && a.size() == 1) siteKey = a.get(0).getAsJsonObject().get("key").getAsString();
                    if (siteKey == null) {
                        LOG.warn("LiveStats: Server nicht gefunden – bitte \"serverKey\" in livestats.json eintragen (z. B. winter oder chill).");
                        return;
                    }
                }
            }
            siteStats = JsonParser.parseString(httpGet(base + "data/" + siteKey + "/stats.json")).getAsJsonObject();
            try {
                siteAdvent = JsonParser.parseString(httpGet(base + "data/advent.json")).getAsJsonObject();
            } catch (Exception ignored) {
            }
            try {
                siteOv = JsonParser.parseString(httpGet(base + "overrides.json")).getAsJsonObject();
            } catch (Exception ignored) {
            }
        } catch (Exception e) {
            LOG.warn("LiveStats: Seitendaten nicht geladen: {}", e.toString());
        }
    }

    static JsonObject obj(JsonObject o, String k) {
        return o != null && o.has(k) && o.get(k).isJsonObject() ? o.getAsJsonObject(k) : null;
    }

    static String num(double v) {
        if (v == Math.rint(v) && Math.abs(v) < 1e15) return String.format(java.util.Locale.GERMAN, "%,d", (long) v);
        return String.format(java.util.Locale.GERMAN, "%,.1f", v);
    }

    static String date(String iso) {
        String[] p = iso.split("-");
        return p.length == 3 ? p[2] + "." + p[1] + "." : iso;
    }

    /** Wird von Compat im Server-Thread aufgerufen; gibt die Chat-Zeilen zurück. */
    static List<String> command(String cmd, String arg, String uuid, String name, Object player) {
        if (cmd.equals("rekord")) return rekord(arg, uuid, name);
        if (cmd.equals("vote")) return vote(arg, uuid, name);
        List<String> out = new ArrayList<>();
        String site = siteUrl();
        if (cmd.equals("seite")) {
            out.add("§b" + (site.isEmpty() ? "Keine Seite eingestellt." : site));
            out.add("§7Befehle: §f/bingo /challenge /serie /advent /vote /rekord");
            return out;
        }
        if (cmd.equals("advent")) return advent(uuid, player);
        JsonObject st = siteStats;
        if (st == null) {
            EX.execute(LiveStats::loadSite);
            out.add("§7Die Daten werden gerade geladen – versuch es in ein paar Sekunden nochmal.");
            return out;
        }
        try {
            switch (cmd) {
                case "bingo" -> bingo(st, uuid, out, site);
                case "challenge" -> challenge(st, uuid, out);
                case "serie" -> serie(st, uuid, out);
                default -> out.add("?");
            }
        } catch (Exception e) {
            LOG.warn("LiveStats: /{} fehlgeschlagen: {}", cmd, e.toString());
            out.add("§cDa ist etwas schiefgelaufen.");
        }
        return out;
    }

    static void bingo(JsonObject st, String uuid, List<String> out, String site) {
        JsonObject b = obj(st, "bingo");
        if (b == null) { out.add("§7Gerade läuft kein Bingo."); return; }
        JsonArray tasks = b.getAsJsonArray("tasks");
        JsonObject prog = obj(b, "progress");
        JsonArray v = prog != null && prog.has(uuid) ? prog.getAsJsonArray(uuid) : null;
        boolean[] done = new boolean[25];
        int cnt = 0;
        List<double[]> open = new ArrayList<>();
        for (int i = 0; i < 25; i++) {
            if (i == 12) { done[i] = true; continue; }
            double t = tasks.get(i).getAsJsonObject().get("target").getAsDouble(), x = v == null ? 0 : v.get(i).getAsDouble();
            done[i] = x >= t;
            if (done[i]) cnt++; else open.add(new double[]{Math.min(x / t, 0.999), i, x, t});
        }
        int lines = 0;
        for (int[] l : LINES) { boolean ok = true; for (int i : l) ok &= done[i]; if (ok) lines++; }
        out.add("§d§lBingo der Woche §r§7(bis " + date(b.get("end").getAsString()) + ")");
        for (int r = 0; r < 5; r++) {
            StringBuilder sb = new StringBuilder("  ");
            for (int c = 0; c < 5; c++) { int i = r * 5 + c; sb.append(i == 12 ? "§e★ " : done[i] ? "§a■ " : "§8■ "); }
            out.add(sb.toString());
        }
        out.add("§fErledigt: §a" + cnt + "/24 §7· Reihen: §e" + lines);
        open.sort((a, z) -> Double.compare(z[0], a[0]));
        if (!open.isEmpty()) out.add("§7Am nächsten dran:");
        for (int k = 0; k < Math.min(3, open.size()); k++) {
            double[] o = open.get(k);
            out.add(" §8- §f" + tasks.get((int) o[1]).getAsJsonObject().get("text").getAsString() + " §7" + num(o[2]) + "/" + num(o[3]));
        }
        if (!site.isEmpty()) out.add("§7Ganze Karte: §b" + site + "#" + siteKey);
    }

    static void challenge(JsonObject st, String uuid, List<String> out) {
        JsonObject c = obj(st, "challenge");
        if (c == null) { out.add("§7Gerade läuft keine Wochen-Challenge."); return; }
        String unit = c.has("unit") ? c.get("unit").getAsString() : "";
        String u = unit.isEmpty() ? "" : unit.equals("×") ? "×" : " " + unit;
        out.add("§6§lWochen-Challenge: §r§f" + c.get("label").getAsString() + " §7(bis " + date(c.get("end").getAsString()) + ")");
        JsonArray rk = c.getAsJsonArray("ranking");
        int me = -1;
        for (int i = 0; i < rk.size(); i++) {
            JsonArray r = rk.get(i).getAsJsonArray();
            if (i < 3) out.add(" §e" + (i + 1) + ". §f" + r.get(1).getAsString() + " §7" + num(r.get(2).getAsDouble()) + u);
            if (r.get(0).getAsString().equals(uuid)) me = i;
        }
        if (rk.isEmpty()) out.add(" §7Noch hat niemand gepunktet – leg los!");
        if (me >= 0) out.add("§fDu: §ePlatz " + (me + 1) + " §7mit " + num(rk.get(me).getAsJsonArray().get(2).getAsDouble()) + u);
        else out.add("§7Du hast diese Woche noch nichts dazu beigetragen.");
    }

    static void serie(JsonObject st, String uuid, List<String> out) {
        JsonObject s = obj(obj(st, "streaks"), uuid);
        if (s == null) { out.add("§7Für dich gibt es noch keine Serie."); return; }
        String today = LocalDate.now(VIENNA).toString(), yest = LocalDate.now(VIENNA).minusDays(1).toString();
        String last = s.has("last") && !s.get("last").isJsonNull() ? s.get("last").getAsString() : "";
        int cur = last.equals(today) || last.equals(yest) ? s.get("cur").getAsInt() : 0;
        out.add("§6§lDeine Serie: §r§f" + cur + (cur == 1 ? " Tag" : " Tage") + " §7(Rekord: " + s.get("best").getAsInt() + ")");
        if (cur > 0 && !last.equals(today)) out.add("§7Heute zählt, sobald du 5 Minuten gespielt hast – die Seite merkt es bei der nächsten Aktualisierung.");
    }

    /** Liest die aktuellen Minecraft-Statistiken eines Spielers direkt aus dem Welt-Ordner (vorher speichern). */
    static JsonObject liveStats(String uuid) {
        try {
            if (server != null) Compat.saveAll(server);
            Path game = FabricLoader.getInstance().getGameDir();
            String level = "world";
            Path props = game.resolve("server.properties");
            if (Files.exists(props)) {
                for (String line : Files.readAllLines(props)) if (line.startsWith("level-name=")) level = line.substring(11).trim();
            }
            for (String sub : new String[]{"stats", "players/stats"}) {
                Path f = game.resolve(level).resolve(sub).resolve(uuid + ".json");
                if (Files.exists(f)) {
                    JsonObject o = obj(JsonParser.parseString(Files.readString(f)).getAsJsonObject(), "stats");
                    return o == null ? new JsonObject() : o;
                }
            }
        } catch (Exception e) {
            LOG.warn("LiveStats: Statistik von {} nicht lesbar: {}", uuid, e.toString());
        }
        return new JsonObject();
    }

    static double taskValue(JsonObject stats, JsonObject task) {
        double sum = 0;
        for (JsonElement pe : task.getAsJsonArray("parts")) {
            JsonObject part = pe.getAsJsonObject();
            JsonObject cat = obj(stats, "minecraft:" + part.get("t").getAsString());
            if (cat == null) continue;
            for (Map.Entry<String, JsonElement> e : cat.entrySet()) {
                String id = e.getKey().startsWith("minecraft:") ? e.getKey().substring(10) : e.getKey();
                boolean ok = false;
                if (part.has("ids")) for (JsonElement x : part.getAsJsonArray("ids")) ok |= x.getAsString().equals(id);
                if (part.has("suffix")) for (JsonElement x : part.getAsJsonArray("suffix")) ok |= id.endsWith(x.getAsString());
                if (ok) sum += e.getValue().getAsDouble();
            }
        }
        return sum / (task.has("div") ? task.get("div").getAsDouble() : 1);
    }

    static synchronized List<String> advent(String uuid, Object player) {
        List<String> out = new ArrayList<>();
        LocalDate d = LocalDate.now(VIENNA);
        if (!cfg.advent) { out.add("§7Der Adventskalender ist auf diesem Server ausgeschaltet."); return out; }
        if (d.getMonthValue() != 12) { out.add("§7Der Adventskalender öffnet am 1. Dezember."); return out; }
        JsonObject a = siteAdvent;
        if (a == null) { EX.execute(LiveStats::loadSite); out.add("§7Der Kalender wird gerade geladen – versuch es gleich nochmal."); return out; }
        if (a.get("year").getAsInt() != d.getYear()) { out.add("§7Der Kalender für dieses Jahr ist noch nicht fertig."); return out; }
        JsonObject players = obj(a, "players"), rewards = obj(a, "rewards");
        JsonArray mine = players != null && players.has(uuid) ? players.getAsJsonArray(uuid) : null, tasks = a.getAsJsonArray("tasks");
        if (mine == null || tasks == null) { out.add("§7Du bist noch nicht im Kalender – er wird bei der nächsten Aktualisierung der Seite ergänzt."); return out; }
        double maxF = a.has("maxFactor") ? a.get("maxFactor").getAsDouble() : 64;
        JsonObject claims;
        try {
            claims = Files.exists(CLAIMS) ? JsonParser.parseString(Files.readString(CLAIMS)).getAsJsonObject() : new JsonObject();
        } catch (Exception e) {
            claims = new JsonObject();
        }
        String ck = d.getYear() + ":" + uuid;
        JsonObject me = claims.has(ck) && claims.get(ck).isJsonObject() ? claims.getAsJsonObject(ck) : new JsonObject();
        if (!me.has("got")) me.add("got", new JsonArray());
        if (!me.has("base")) me.add("base", new JsonObject());
        JsonArray got = me.getAsJsonArray("got");
        JsonObject base = me.getAsJsonObject("base");
        java.util.Set<Integer> have = new java.util.HashSet<>();
        got.forEach(x -> have.add(x.getAsInt()));
        JsonObject stats = liveStats(uuid);
        int today = Math.min(d.getDayOfMonth(), 24), gave = 0, opened = 0;
        out.add("§6§l❄ Adventskalender ❄");
        for (int day = 1; day <= today; day++) {
            if (have.contains(day)) continue;
            JsonObject task = tasks.get(day - 1).getAsJsonObject(), r = obj(rewards, mine.get(day - 1).getAsString());
            if (r == null) continue;
            int late = Math.max(0, d.getDayOfMonth() - day);
            double factor = Math.min(Math.pow(2, late), maxF), need = task.get("target").getAsDouble() * factor;
            String text = task.get("text").getAsString().replace("{n}", num(need)), extra = late > 0 ? " §8(nachgeholt: " + num(factor) + "×)" : "";
            double now = taskValue(stats, task);
            String bk = String.valueOf(day);
            if (!base.has(bk)) {
                base.addProperty(bk, now);
                opened++;
                out.add("§e☐ Türchen " + day + ": §f" + text + extra + " §7– ab jetzt zählt's!");
                continue;
            }
            double prog = now - base.get(bk).getAsDouble();
            if (prog + 1e-9 < need) {
                out.add("§e☐ Türchen " + day + ": §f" + text + extra + " §7– " + num(Math.max(0, prog)) + "/" + num(need));
                continue;
            }
            String tier = r.get("tier").getAsString(), col = tier.equals("final") ? "§6" : tier.equals("gut") ? "§d" : "§b";
            List<String> lore = new ArrayList<>(List.of("§7" + r.get("lore").getAsString(), "§8Adventskalender " + d.getYear() + " · Türchen " + day));
            boolean fw = r.has("firework") && r.get("firework").getAsBoolean();
            if (!Compat.give(player, r.get("item").getAsString(), r.get("count").getAsInt(), "§r" + col + r.get("name").getAsString(), lore, fw)) {
                LOG.warn("LiveStats: unbekanntes Item {} im Adventskalender", r.get("item").getAsString());
                continue;
            }
            got.add(day);
            gave++;
            out.add("§a✔ Türchen " + day + " geschafft: " + col + r.get("count").getAsInt() + "× " + r.get("name").getAsString());
        }
        claims.add(ck, me);
        try {
            Files.writeString(CLAIMS, GSON.toJson(claims));
        } catch (Exception e) {
            LOG.warn("LiveStats: Adventskalender nicht gespeichert: {}", e.toString());
        }
        if (out.size() == 1) out.add(have.size() >= 24 ? "§7Du hast alle 24 Türchen geschafft – frohe Weihnachten!" : "§7Alles erledigt! Das nächste Türchen gibt es morgen.");
        else if (gave == 0 && opened == 0) out.add("§7Erledige die Aufgaben und tippe dann wieder §f/advent§7.");
        else if (opened > 0) out.add("§7Erledige die Aufgabe und tippe dann wieder §f/advent§7, um dein Geschenk zu holen.");
        return out;
    }

    static int recDay() {
        return (int) (java.time.temporal.ChronoUnit.DAYS.between(LocalDate.of(2026, 1, 1), LocalDate.now(VIENNA)) & 4095);
    }

    static synchronized List<String> rekord(String arg, String uuid, String name) {
        List<String> out = new ArrayList<>();
        String c = arg == null ? "" : arg.toUpperCase().replaceAll("[^A-Z0-9]", "");
        if (c.length() != 12) {
            out.add("§7Spiel ein Minispiel auf der Seite – am Ende bekommst du einen Code. Den tippst du hier ein: §f/rekord ABCD-EFGH-JKLM");
            return out;
        }
        long v = 0;
        for (char ch : c.toCharArray()) {
            int i = RA.indexOf(ch);
            if (i < 0) { out.add("§cDieser Code ist ungültig."); return out; }
            v = v * 32 + i;
        }
        int chk = (int) (v & 4095), score = (int) ((v >>> 12) & ((1L << 30) - 1)), day = (int) ((v >>> 42) & 4095), g = (int) ((v >>> 54) & 63);
        int h = (int) ((long) (g * 131 + day) * 1000003L + score);
        h = (h ^ 0x5A5A5) * 0x9E3779B1;
        if ((h & 4095) != chk || g >= REC_NAMES.length) { out.add("§cDieser Code ist ungültig – bitte genau abtippen."); return out; }
        int age = (recDay() - day + 4096) % 4096;
        if (age > 2) { out.add("§cDieser Code ist abgelaufen – Codes gelten 2 Tage."); return out; }
        JsonObject m = modData();
        JsonObject mine = sub(sub(m, "records"), uuid);
        String gk = String.valueOf(g);
        JsonArray arr = mine.has(gk) ? mine.getAsJsonArray(gk) : new JsonArray();
        for (JsonElement e : arr) {
            JsonArray x = e.getAsJsonArray();
            if (x.get(0).getAsInt() == score && x.get(1).getAsInt() == day) { out.add("§7Diesen Rekord hast du schon eingetragen."); return out; }
        }
        JsonArray entry = new JsonArray();
        entry.add(score);
        entry.add(day);
        arr.add(entry);
        while (arr.size() > 20) arr.remove(0);
        mine.add(gk, arr);
        sub(m, "names").addProperty(uuid, name);
        saveModData(m);
        requestRefresh();
        out.add("§a✔ Rekord eingetragen: §f" + REC_NAMES[g] + " §7– §e" + num(score));
        out.add(cfg.githubToken == null || cfg.githubToken.isBlank() ? "§7(Auf der Seite erscheint er, sobald in livestats.json ein githubToken eingetragen ist.)"
                : "§7Er erscheint in ein paar Minuten auf der Seite.");
        return out;
    }

    static JsonObject activePoll() {
        JsonObject ov = siteOv;
        if (ov == null || !ov.has("polls") || !ov.get("polls").isJsonArray()) return null;
        String today = LocalDate.now(VIENNA).toString();
        JsonObject best = null;
        for (JsonElement e : ov.getAsJsonArray("polls")) {
            JsonObject p = e.getAsJsonObject();
            String end = p.has("end") && !p.get("end").isJsonNull() ? p.get("end").getAsString() : "";
            if (!end.isEmpty() && end.compareTo(today) < 0) continue;
            if (best == null || p.get("id").getAsString().compareTo(best.get("id").getAsString()) > 0) best = p;
        }
        return best;
    }

    static synchronized List<String> vote(String arg, String uuid, String name) {
        List<String> out = new ArrayList<>();
        JsonObject p = activePoll();
        if (p == null) {
            if (siteOv == null) EX.execute(LiveStats::loadSite);
            out.add("§7Gerade läuft keine Umfrage.");
            return out;
        }
        String pid = p.get("id").getAsString();
        JsonArray opts = p.getAsJsonArray("opts");
        JsonObject m = modData();
        JsonObject votes = sub(sub(m, "votes"), pid);
        int mine = votes.has(uuid) ? votes.get(uuid).getAsInt() : 0;
        String a = arg == null ? "" : arg.trim();
        if (a.matches("\\d+")) {
            int n = Integer.parseInt(a);
            if (n < 1 || n > opts.size()) { out.add("§cBitte eine Zahl von 1 bis " + opts.size() + "."); return out; }
            votes.addProperty(uuid, n);
            sub(m, "names").addProperty(uuid, name);
            saveModData(m);
            requestRefresh();
            out.add("§a✔ Deine Stimme: §f" + opts.get(n - 1).getAsString() + (mine > 0 && mine != n ? " §7(geändert)" : ""));
            out.add("§7Das Ergebnis siehst du auf der Seite.");
            return out;
        }
        out.add("§b§lUmfrage: §r§f" + p.get("q").getAsString());
        for (int i = 0; i < opts.size(); i++) out.add(" §e" + (i + 1) + " §f" + opts.get(i).getAsString() + (mine == i + 1 ? " §a← deine Stimme" : ""));
        out.add("§7Abstimmen mit §f/vote <Nummer>§7, z. B. §f/vote 1");
        return out;
    }

    /** Fasst Meldungen der letzten Sekunden zu einer Nachricht zusammen (schont das Discord-Limit). */
    static void tick() {
        try {
            if (OUT.isEmpty() || cfg.discordWebhook == null || cfg.discordWebhook.isBlank()) return;
            StringBuilder sb = new StringBuilder();
            String m;
            while ((m = OUT.peek()) != null && sb.length() + m.length() + 1 < 1900) {
                sb.append(OUT.poll()).append('\n');
            }
            if (sb.length() == 0) return;
            JsonObject body = new JsonObject();
            body.addProperty("content", sb.toString().trim());
            JsonObject allowed = new JsonObject();
            allowed.add("parse", new JsonArray());
            body.add("allowed_mentions", allowed);
            send(cfg.discordWebhook, body.toString(), null, "Discord");
        } catch (Throwable t) {
            LOG.warn("Discord-Versand: {}", t.toString());
        }
    }

    static String esc(String s) {
        StringBuilder b = new StringBuilder();
        for (char c : s.toCharArray()) {
            if ("\\_*~`|>".indexOf(c) >= 0) b.append('\\');
            b.append(c);
        }
        return b.toString();
    }

    static void countDiamond(String name) {
        synchronized (LOCK) {
            DIAMONDS.merge(name, 1, Integer::sum);
            if (!diamondFlushPending) {
                diamondFlushPending = true;
                EX.schedule(LiveStats::flushDiamonds, Math.max(5, cfg.diamondBatchSeconds), TimeUnit.SECONDS);
            }
        }
    }

    static void flushDiamonds() {
        Map<String, Integer> snap;
        synchronized (LOCK) {
            snap = new HashMap<>(DIAMONDS);
            DIAMONDS.clear();
            diamondFlushPending = false;
        }
        snap.forEach((n, c) -> say("💎 **" + esc(n) + "** hat " + c + (c == 1 ? " Diamanterz" : " Diamanterze") + " abgebaut."));
    }

    // ---------- Statistikseite aktualisieren ----------
    static void requestRefresh() {
        if (cfg.githubToken == null || cfg.githubToken.isBlank() || cfg.githubRepo == null || cfg.githubRepo.isBlank()) return;
        synchronized (LOCK) {
            if (refreshPending) return;
            refreshPending = true;
            long wait = Math.max(0, lastRefresh + cfg.refreshMinSeconds * 1000L - System.currentTimeMillis());
            EX.schedule(LiveStats::doRefresh, wait, TimeUnit.MILLISECONDS);
        }
    }

    static void doRefresh() {
        synchronized (LOCK) {
            refreshPending = false;
            lastRefresh = System.currentTimeMillis();
        }
        MinecraftServer s = server;
        if (s == null) {
            dispatch();
            return;
        }
        // Statistik-Dateien erst speichern lassen (Server-Thread), dann den Workflow starten
        s.execute(() -> {
            try {
                Compat.saveAll(s);
            } catch (Throwable t) {
                LOG.warn("Spielerdaten konnten nicht gespeichert werden: {}", t.toString());
            }
            EX.schedule(LiveStats::dispatch, 4, TimeUnit.SECONDS);
        });
    }

    static void dispatch() {
        try {
            String url = "https://api.github.com/repos/" + cfg.githubRepo + "/actions/workflows/" + cfg.githubWorkflow + "/dispatches";
            Map<String, String> h = new HashMap<>();
            h.put("Authorization", "Bearer " + cfg.githubToken.trim());
            h.put("Accept", "application/vnd.github+json");
            h.put("X-GitHub-Api-Version", "2022-11-28");
            JsonObject body = new JsonObject();
            body.addProperty("ref", cfg.githubRef);
            JsonObject md = modData();
            if (siteKey != null && (md.has("records") || md.has("votes"))) {
                md.addProperty("server", siteKey);
                JsonObject in = new JsonObject();
                in.addProperty("moddata", md.toString());
                body.add("inputs", in);
            }
            send(url, body.toString(), h, "GitHub");
        } catch (Throwable t) {
            LOG.warn("GitHub-Aufruf: {}", t.toString());
        }
    }

    static void periodic() {
        try {
            if (ONLINE.get() > 0 && System.currentTimeMillis() - lastRefresh >= cfg.refreshEverySeconds * 1000L) requestRefresh();
        } catch (Throwable t) {
            LOG.warn("livestats periodic: {}", t.toString());
        }
    }

    // ---------- HTTP ----------
    static void send(String url, String body, Map<String, String> headers, String label) {
        HttpRequest.Builder b = HttpRequest.newBuilder(URI.create(url))
                .timeout(Duration.ofSeconds(15))
                .header("Content-Type", "application/json")
                .header("User-Agent", "livestats-mod");
        if (headers != null) headers.forEach(b::header);
        HTTP.sendAsync(b.POST(HttpRequest.BodyPublishers.ofString(body)).build(), HttpResponse.BodyHandlers.ofString())
                .thenAccept(r -> {
                    if (r.statusCode() >= 300) LOG.warn("{}: HTTP {} {}", label, r.statusCode(), r.body());
                })
                .exceptionally(e -> {
                    LOG.warn("{}: {}", label, e.toString());
                    return null;
                });
    }

        st.set(DataComponentTypes.LORE, new LoreComponent(lore.stream().map(l -> (Text) Text.literal(l)).toList()));
        ServerPlayerEntity p = (ServerPlayerEntity) player;
        if (!p.getInventory().insertStack(st)) p.dropItem(st, false);
        return true;
    }
